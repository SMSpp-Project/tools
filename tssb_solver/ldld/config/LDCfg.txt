ComputeConfig         # exact type of the ComputeConfig object

1  # f_diff == 1 ==> only the entries listed below are changed

16 # number of integer parameters

# now all the integer parameters
intPushCostToOwner     1 # whether sub-Block Objective are changed
intDoEasy              0 # do easy components for (some) LagBFunction
intMaxIter         10000 # max number of iterations for each call
intMaxThread           1 # max number of new tasks to spawn
intLogVerb             0 # log verbosity of main Bundle algorithm
intWZNorm              2 # norm used in the norm-based stopping condition
intBPar1              20 # discard items useless for <this> iterations
intBPar2             100 # max bundle size per component
intBPar3               1 # max n. of items fetched from Fi() per iteration
intBPar4               1 # min n. of items fetched from Fi() per iteration
intBPar6               0 # second parameter for the dynamic max n. of items
intBPar7              11 # how to deal with the global pools
intMnSSC               0 # min n. of consecutive SS with the same t to raise
                         # t
intMnNSC               1 # min n. of consecutive NS with the same t to cut t
inttSPar1             12 # long-term t-strategy (0 = none, 4 = soft,
                         # 8 = hard, 12 = balancing)
intMaxNrEvls           2 # max number of function evaluations per iteration

15 # number of double parameters

# now all the double parameters
dblRelAcc  1e-6 # relative accuracy required to solution
dblNZEps   1e-2 # threshold to declare 0 the || residual ||
dbltStar  -1e+3 # multiplied to || residual || to estimate the gap
dblBPar5      4 # first parameter for the dynamic max n. of items
dblm1     -0.30 # a NS is possible if Fi( L1 ) >= Fi( L ) + | m1 | v*
dblm2      0.90 # a SS is possible if Fi( L1 ) <= Fi( L ) + (1 - m2) v*
dblm3      0.99 # a NR is computed if sigma* < - t * m3 * || z* ||
dblmxIncr    10 # max increase of t
dblmnIncr   1.5 # min increase of t (each time it is increased)
dblmxDecr   0.1 # max decrease of t
dblmnDecr  0.66 # min decrease of t (each time it is decreased)
dbltMaior  1e+6 # maximum value for t
dbltMinor 1e-10 # minimum value for t
dbltInit   1e+1 # initial value for t
dbltSPar2  1e-2 # parameter for the long-term t-strategy

2 # number of string parameters

# now all the string parameters
str_LDSlv_ISName BundleSolver # the inner Solver
strMPBSolverCfg    MPBCfg.txt # BlockSolverConfig of the master

# [none]
