# A txt description of a BlockConfig for a ThermalUnitBlock that is being
# solved by a :MILPSolver, or by its own Solver under a
# LagrangianDualSolver. As such it is relevant:
#
# - if dynamic constraints (Perspective Cuts) are separated, and what are
#   corresponding tolerances, as controlled by the Configuration of
#   dynamic constraints
#
# - which of the several different available formulation is selected, as
#   controlled by the Configuration of static variables
#
# - which reserves are in the Objective also when their cost is 0, as
#   controlled by the Configuration of the objective
#
# Note that if the static variables do not include P/C (+8), then the
# dynamic constraints are actually irrelevant
#
# - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - -
# - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - -

BlockConfig # exact type of the ComputeConfig object

1  # the BlockConfig is a "differential" one

2  # version of the BlockConfig format

# the Configuration for the structure, i.e., the tree of sub-Block
*  # [none]

# static_constraints
*    # [none]

# dynamic_constraints
SimpleConfiguration<std::pair<double,double>>
1e-6    # threshold parameter for P/C separation
1e-4    # tolerance value to consider a binary variable

# static_variables
SimpleConfiguration<int>
1    # 0 = 3bin, 1 = T, 2 = pt, 3 = DP, 4 = SU, 5 = SD, 6 = SUSD
     # (Formulation)
#9    # +8 with Perspective Cuts
#17   # +16: start-up / shut-down variables are continuous

# dynamic_variables
*    # [none]

# objective
SimpleConfiguration<int>
3    # the reserve terms in the Objective also with cost 0: +1 primary,
     # +2 secondary (a LagrangianDualSolver prices them there)

# is_feasible
*    # [none]

# is_optimal
*    # [none]

# solution
*    # [none]

# extra
*    # [none]
